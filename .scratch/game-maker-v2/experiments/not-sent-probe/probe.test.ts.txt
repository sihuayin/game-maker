/** 一份**逐字节固定**的回应体（200 那一路）。 */
const raw = (body: string, contentType: string): typeof fetch =>
  (async () => new Response(body, { status: 200, headers: { "content-type": contentType } })) as unknown as typeof fetch;

type Landing = {
  threw: boolean; name: string; kind: string | null; exitCode: number | null;
  ledger: unknown; message: string;
};
const land = async (p: Promise<unknown>): Promise<Landing> => {
  try {
    await p;
    return { threw: false, name: "(没抛)", kind: null, exitCode: null, ledger: null, message: "" };
  } catch (e) {
    const err = e as Error;
    const kind = (err as unknown as { kind?: string }).kind ?? null;
    return {
      threw: true, name: err.name ?? err.constructor.name, kind, exitCode: exitCodeOfError(err),
      ledger: ledgerOfError(err) ?? null, message: err.message.slice(0, 90)
    };
  }
};

/** 三个命令各跑一发同样的回应体。⚠️ `planAssets` / `compileRuntime` 走 R16 工具那条路，`compileTdGame` 走旧文本那条。 */
const three = async (body: string, contentType: string): Promise<Record<string, Landing>> => {
  const d = tmp();
  const f = raw(body, contentType);
  return {
    planAssets: await land(planAssets({
      designPath: path.join(d, "game-design.json"), visualWorldPath: path.join(d, "visual-world.json"),
      outRoot: path.join(d, "out"), transport: { baseUrl: "http://x", apiKey: "k" }, fetchImpl: f
    })),
    compileRuntime: await land(compileRuntime({
      designPath: path.join(d, "game-design.json"), packDir: PACK_RUNTIME,
      outRoot: path.join(d, "out"), transport: { baseUrl: "http://x", apiKey: "k" }, fetchImpl: f
    })),
    compileTdGame: await land(compileTdGame({
      requirementPath: path.join(d, "requirement.md"), packDir: PACK_TD,
      outRoot: path.join(d, "out"), transport: { baseUrl: "http://x", apiKey: "k" }, fetchImpl: f
    }))
  };
};

const BODIES: [string, string, string][] = [
  ["① 200 + text/plain 说人话（不是 JSON）", "上游今天心情不好，请稍后再试", "text/plain"],
  ["② 200 + JSON 字面量 null", "null", "application/json"],
  ["③ 200 + JSON 空数组 []", "[]", "application/json"],
  ["④ 200 + JSON 数字 123", "123", "application/json"],
  ["⑤ 200 + {\"error\":{\"message\":…}}（代理把 500 转成 200）", JSON.stringify({ error: { message: "upstream returned 500" } }), "application/json"],
  ["⑥ 200 + 合法 messages 形状但 content 全是非文本块", JSON.stringify({ content: [{ type: "tool_use", name: "x", input: {} }] }), "application/json"],
  ["⑦ 200 + content 不是数组", JSON.stringify({ content: 5 }), "application/json"]
];

describe("票 51 探针", () => {
  it("量 1 · 2 · 4：同一份回应体走三个命令", async () => {
    const report: Record<string, Record<string, Landing>> = {};
    for (const [label, body, ct] of BODIES) report[label] = await three(body, ct);
    fs.writeFileSync(path.join(process.env.PROBE_OUT ?? os.tmpdir(), "probe51-three.json"), JSON.stringify(report, null, 2));
    console.log("\n=== 量 1/2/4：三个命令 × 六种 200 回应体 ===\n" + JSON.stringify(report, null, 2));
  }, 60_000);

  it("量 3：`trip({})` 之后 `record()` 的逐字段读数（真 `spentCall`，逐字复制、只加了 export）", () => {
    const a = spentCall("compile-td-game", "td-config");
    a.trip({});
    const phantom = a.record();
    const b = spentCall("compile-td-game", "td-config");
    const empty = b.record();
    // 对照：一次真往返（成功）
    const c = spentCall("compile-td-game", "td-config");
    c.trip({ spent: { model: "deepseek-v4-pro", usage: { outputTokens: 123 } } });
    const real = c.record();
    const out = { "trip({}) 之后的 record()": phantom, "一次都没 trip 的 record()": empty, "一次真往返的 record()": real };
    fs.writeFileSync(path.join(process.env.PROBE_OUT ?? os.tmpdir(), "probe51-record.json"), JSON.stringify(out, null, 2));
    console.log("\n=== 量 3：record() 逐字段 ===\n" + JSON.stringify(out, null, 2));
  });

  it("量 3b：陷阱演练 —— 把 `not-sent` 那一行照抄成带账，账里出现什么（真链路，只改了这一行）", async () => {
    const d = tmp();
    const e = await patched.compileTdGame({
      requirementPath: path.join(d, "requirement.md"), packDir: PACK_TD,
      outRoot: path.join(d, "out"), transport: { baseUrl: "", apiKey: "" }
    }).catch((x: unknown) => x as Error);
    const kind = (e as unknown as { kind?: string }).kind ?? null;
    const out = {
      "空凭据那条真 not-sent": { name: e.name, kind, exitCode: exitCodeOfError(e), message: e.message.slice(0, 80) },
      "接上账之后 ledgerOfError(e)": ledgerOfError(e) ?? null
    };
    fs.writeFileSync(path.join(process.env.PROBE_OUT ?? os.tmpdir(), "probe51-phantom.json"), JSON.stringify(out, null, 2));
    console.log("\n=== 量 3b：假账演练 ===\n" + JSON.stringify(out, null, 2));
  }, 60_000);

  it("量 5：空凭据（真 not-sent）走三个命令各自的对外落点", async () => {
    const d = tmp();
    const out = {
      planAssets: await land(planAssets({ designPath: path.join(d, "game-design.json"), visualWorldPath: path.join(d, "visual-world.json"),
        outRoot: path.join(d, "out"), transport: { baseUrl: "", apiKey: "" } })),
      compileRuntime: await land(compileRuntime({ designPath: path.join(d, "game-design.json"), packDir: PACK_RUNTIME,
        outRoot: path.join(d, "out"), transport: { baseUrl: "", apiKey: "" } })),
      compileTdGame: await land(compileTdGame({ requirementPath: path.join(d, "requirement.md"), packDir: PACK_TD,
        outRoot: path.join(d, "out"), transport: { baseUrl: "", apiKey: "" } }))
    };
    fs.writeFileSync(path.join(process.env.PROBE_OUT ?? os.tmpdir(), "probe51-credentials.json"), JSON.stringify(out, null, 2));
    console.log("\n=== 量 5：空凭据 ===\n" + JSON.stringify(out, null, 2));
  }, 60_000);
});
